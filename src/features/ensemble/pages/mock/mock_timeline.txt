const mmss = (t) => { t = Math.max(0, Math.floor(t + 1e-6)); return String(Math.floor(t / 60)).padStart(2, '0') + ':' + String(t % 60).padStart(2, '0'); };

===== 3. WAVEFORM + THƯỚC + KÉO/TUA (trong class Component) =====
  wfPath(i, zoom) {
    const key = i + '|' + zoom + '|' + (i === 4 ? this.eng.userVer : 0); if (this.wf[key]) return this.wf[key];
    const data = this.dataOf(i); const W = DUR * PPS; const pitch = 2.8; const n = Math.floor(W / pitch); const per = Math.floor(data.length / n);
    if (this.maxs[i === 4 ? this.state.part : i] == null) { const src = this.data[i === 4 ? this.state.part : i]; let m = 0; for (let j = 0; j < src.length; j += 5) { const a = Math.abs(src[j]); if (a > m) m = a; } this.maxs[i === 4 ? this.state.part : i] = m || 1; }
    const mx = this.maxs[i === 4 ? this.state.part : i]; let d = '';
    for (let c = 0; c < n; c++) {
      let m = 0; for (let j = c * per; j < (c + 1) * per; j += 5) { const a = Math.abs(data[j]); if (a > m) m = a; }
      const h = Math.max(1.1, Math.pow(Math.min(1, m / mx), 0.75) * 22); const x = (c * pitch + pitch / 2).toFixed(1);
      d += 'M' + x + ' ' + (28 - h).toFixed(1) + 'V' + (28 + h).toFixed(1);
    }
    this.wf[key] = d; return d;
  }
  ticksFor(zoom) {
    if (this.tickCache[zoom]) return this.tickCache[zoom]; const pps = PPS; const ticks = []; const labels = [];
    for (let t = 0; t <= DUR; t++) {
      const major = t % 15 === 0; const minor = !major && t % 5 === 0; const micro = !major && !minor; if (!major && !minor && !micro) continue;
      const h = major ? 12 : minor ? 8 : 4; ticks.push({ style: 'position:absolute;bottom:0;left:' + (LEAD + t * pps).toFixed(1) + 'px;width:1px;height:' + h + 'px;background:' + (major ? '#7C8C82' : '#B5C0B8') + ';' });
      if (major) labels.push({ text: mmss(t), style: 'position:absolute;top:5px;left:' + (LEAD + t * pps + (t === 0 ? 5 : 4)).toFixed(1) + 'px;font-family:\'IBM Plex Mono\',monospace;font-size:11px;font-weight:600;color:' + (t === 0 ? '#145B43' : '#5E6F66') + ';' });
    }
    return (this.tickCache[zoom] = { ticks, labels });
  }

  // ---------- Pointer: kéo/chạm để seek ----------
  xToT(e) { const el = document.getElementById('ens-content'); const r = el.getBoundingClientRect(); return (e.clientX - r.left - LEAD) / PPS; }
  onDown(e) {
    const s = this.eng; if (s.rec === 'countin' || s.rec === 'recording') return;
    s.tap = { x: e.clientX, y: e.clientY, p0: this.curPos(), moved: false }; s.wasPlaying = s.playing;
    try { e.currentTarget.setPointerCapture(e.pointerId); } catch (x) { /* bỏ qua */ }
  }
  onMove(e) {
    const s = this.eng; const g = s.tap; if (!g) return; const dx = e.clientX - g.x;
    if (!g.moved && Math.abs(dx) < 5) return;
    if (!g.moved) { g.moved = true; s.drag = true; if (s.playing) { s.pos = this.curPos(); this.stopSources(); s.playing = false; this.sync(); } }
    s.pos = Math.max(0, Math.min(DUR, g.p0 - dx / PPS)); this.setState({ pos: s.pos });
  }
  onUp(e) {
    const s = this.eng; const g = s.tap; s.tap = null; if (!g) return;
    if (g.moved) { s.drag = false; if (s.wasPlaying) this.play(); else this.sync(); }
    else this.seek(this.xToT(e));
  }
  onCancel() { const s = this.eng; const g = s.tap; s.tap = null; if (g && g.moved) { s.drag = false; if (s.wasPlaying) this.play(); else this.sync(); } }

  renderVals() {

    const zoom = 0; const pps = PPS; const W = DUR * PPS; const vw = this.viewW(); const WC = Math.round(LEAD + W + Math.max(LEAD, vw / 2)); const pos = e.playing ? s.pos : s.pos;
    const rec = s.rec; const idle = rec === 'idle', isCount = rec === 'countin', isRec = rec === 'recording', isPreview = rec === 'preview';
    const part = s.part; const hasUser = !!this.user && (isCount || isRec || isPreview);
    const nRows = hasUser ? 5 : 4; const cardH = RULER_H + 2 + nRows * (ROW_H + ROW_GAP) + 2;
    const parts = TRACKS.slice(0, 4).map((t, i) => ({ name: t.name, on: part === i,
      pick: () => { if (isCount || isRec) return; if (isPreview) this.retake(); e.muted = { [i]: true }; this.applyGains(); this.setState({ part: i, muted: { [i]: true } }); },
      style: 'flex:1 1 0;min-height:44px;border-radius:14px;font-size:13.5px;font-weight:' + (part === i ? 700 : 600) + ';cursor:' + ((isCount || isRec) ? 'not-allowed' : 'pointer') + ';padding:0 4px;' + (part === i ? 'border:2px solid #145B43;background:#145B43;color:#FFFFFF;' : 'border:1px solid #CFDCD3;background:#FFFFFF;color:#123F33;') + ((isCount || isRec) && part !== i ? 'opacity:.5;' : '') }));
    const rows = [];
    for (let i = 0; i < nRows; i++) {

      const played = Math.max(0, Math.min(W, pos * pps));

      rows.push({

        toggleMute: () => { if (i === 4) return; e.muted = Object.assign({}, e.muted, { [i]: !e.muted[i] }); this.applyGains(); this.setState({ muted: Object.assign({}, e.muted) }); },
        labelStyle: 'width:100%;height:' + ROW_H + 'px;margin-bottom:' + ROW_GAP + 'px;border:none;border-top:1px solid rgba(250,248,239,0.12);background:transparent;display:flex;align-items:center;gap:8px;padding:0 8px;box-sizing:border-box;cursor:pointer;color:#FFFFFF;text-align:left;' + (muted ? 'opacity:.5;' : '') + (mine && i < 4 ? 'box-shadow:inset 3px 0 0 #C8A63C;' : ''),

        rowStyle: 'position:absolute;left:' + LEAD + 'px;top:' + top + 'px;width:' + W + 'px;height:' + ROW_H + 'px;border-radius:12px;overflow:hidden;background:' + t.bg + ';' + (mine && i < 4 ? 'box-shadow:inset 0 0 0 1.5px #C8A63C;' : '') + (muted ? 'opacity:.55;' : ''),
        showLight, colLight: t.light, colDark: t.dark,
        pathD: ready ? this.wfPath(i, zoom) : '',
        playedStyle: 'position:absolute;left:0;top:0;bottom:0;width:' + played.toFixed(1) + 'px;overflow:hidden;',
      });
    }
    const tk = this.ticksFor(zoom); const px = LEAD + Math.max(0, Math.min(W, pos * pps)); const sl = Math.max(0, Math.min(WC - vw, px - vw / 2));
    const canTransport = ready && !isCount && !isRec;

'#123F33' : '#93A399') + ';border-radius:14px;padding:0 12px;font-size:13.5px;font-weight:600;display:inline-flex;align-items:center;gap:6px;cursor:' + (on ? 'pointer' : 'not-allowed') + ';' + (on ? '' : 'opacity:.6;');

      zoomOut: () => { if (zoom > 0) { const sc = document.getElementById('ens-scroll'); this.setState({ zoom: zoom - 1 }); if (sc) sc.scrollLeft = 0; } },
